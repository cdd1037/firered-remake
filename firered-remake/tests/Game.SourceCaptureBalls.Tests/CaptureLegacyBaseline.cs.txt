using System.Collections.Immutable;
using System.Text.Json;
using Game.Content;
using Game.Domain.Battle;
using Game.Domain.Capture;
using Game.Domain.Pokemon;
using Game.Domain.Randomness;
using Game.Domain.WildBattle;
if (args.Length != 3 || args[0] != "--capture") throw new ArgumentException("--capture CONTENT_ROOT OUTPUT");
if (File.Exists(args[2])) throw new IOException("Refuse to overwrite frozen full results");
if (typeof(CaptureRules).GetField("ConditionalBallsRulesVersion") is not null) throw new InvalidOperationException("Capture requires the unchanged pre-C14 Domain");
var results = new List<object>();
foreach (CaptureBall ball in new[] { CaptureBall.Master, CaptureBall.Ultra, CaptureBall.Great, CaptureBall.Poke })
foreach (CaptureStatus status in Enum.GetValues<CaptureStatus>())
foreach (int rate in new[] { 1, 3, 45, 127, 255 })
foreach (int hp in new[] { 1, 101 })
foreach (uint seed in new uint[] { 0, 31, 0xffffffff })
{
    var input = new CaptureInput(101, hp, rate, ball, status);
    results.Add(new { kind = "core", input, result = CaptureRules.Resolve(input, new(seed, 7)) });
}
var content = SourceMechanicsCatalogLoader.LoadAuthenticated(args[1]); var catalog = content.Catalog;
WildBattleCombatant Mon(string species, string move) {
 var p = SourcePokemonFactory.Create(content, new(new("pret:SPECIES_"+species), 40, 42, IvPolicy.Fixed(31), PersonalityPolicy.Fixed(0)), new(0)).Pokemon;
 var id = new Game.Domain.Ids.MoveId("pret:MOVE_"+move); int pp=catalog.Moves[id].Pp;
 return new(p with { Moves = [new(id,pp,pp)] });
}
foreach (CaptureBall ball in new[] { CaptureBall.Poke, CaptureBall.Great })
foreach (string species in new[] { "RATTATA", "MAGIKARP", "SNORLAX", "DITTO" })
foreach (uint seed in new uint[] { 0, 31, 42, 0xffffffff })
{
 var start = WildBattleRules.StartSourceWild(content, BattleRulesProfile.CanonicalSourceSingles, [Mon("SNORLAX","DEFENSE_CURL")],0,Mon(species,"DEFENSE_CURL"),new(seed));
 if(!start.Succeeded) throw new Exception(start.Fault);
 var result = WildBattleRules.ResolveBall(catalog,start.State!,ball);
 if(!result.Succeeded) throw new Exception(result.Fault);
 results.Add(new {kind="wild",ball,state=start.State,result});
}
File.WriteAllText(args[2],JsonSerializer.Serialize(new{sourceBase="e26aec07c73ac5f60cc55e7d9d93bc5698bacffe",results})+"\n");
Console.WriteLine($"CAPTURED {results.Count} full results");
